CREATE TABLE kituser (
	id	 BIGSERIAL,
	user_name VARCHAR(512) NOT NULL,
	password	 VARCHAR(512) NOT NULL,
	PRIMARY KEY(id)
);

CREATE TABLE kit (
	id	 BIGSERIAL,
	kit_name	 VARCHAR(512),
	payed	 BOOL NOT NULL DEFAULT false,
	kituser_id BIGINT NOT NULL,
	PRIMARY KEY(id)
);

CREATE TABLE category (
	id		 BIGSERIAL,
	category_name VARCHAR(512),
	kituser_id	 BIGINT NOT NULL,
	PRIMARY KEY(id)
);

CREATE TABLE item (
	id		 BIGSERIAL,
	item_name	 VARCHAR(512) NOT NULL,
	category_id BIGINT NOT NULL,
	kit_id	 BIGINT NOT NULL,
	kituser_id	 BIGINT NOT NULL,
	PRIMARY KEY(id)
);

ALTER TABLE item ADD CONSTRAINT item_fk1 FOREIGN KEY (category_id) REFERENCES category(id) ON DELETE CASCADE;
