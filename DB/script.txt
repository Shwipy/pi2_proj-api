CREATE TABLE users (
	id	 BIGSERIAL,
	user_name VARCHAR(512) NOT NULL,
	password	 BOOL NOT NULL,
	PRIMARY KEY(id)
);

CREATE TABLE kit (
	id	 BIGSERIAL,
	kit_name VARCHAR(512),
	payed	 BOOL NOT NULL DEFAULT false,
	users_id BIGINT NOT NULL,
	PRIMARY KEY(id)
);

CREATE TABLE category (
	id		 BIGSERIAL,
	category_name VARCHAR(512) NOT NULL,
	item_id	 BIGINT NOT NULL,
	users_id	 BIGINT NOT NULL,
	PRIMARY KEY(id)
);

CREATE TABLE item (
	id	 BIGSERIAL,
	item_name VARCHAR(512) NOT NULL,
	kit_id	 BIGINT NOT NULL,
	users_id	 BIGINT NOT NULL,
	PRIMARY KEY(id)
);
